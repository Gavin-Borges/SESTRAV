# uv dependency override file.
#
# Forces setuptools to resolve to >=83.0.0 (closes GHSA-h35f-9h28-mq5c)
# regardless of torch's declared `setuptools<82` build-metadata cap. This is
# runtime-safe: torch 2.12.0 imports and runs fine against setuptools 83, and
# every install path in this repo pins by hash (`pip install --require-hashes
# --no-deps`), which never re-resolves or enforces torch's metadata cap.
#
# Without this override, `uv pip compile` (and any resolving installer) fails
# with ResolutionImpossible on requirements.in / environments/requirements-lock.in,
# because both declare a setuptools floor >=83.0.0 that conflicts with torch's
# <82 cap. See environments/requirements-lock.in for the longer version of this
# rationale.
#
# Regenerate the lockfiles with this override applied:
#   uv pip compile --generate-hashes --overrides overrides.txt \
#     --python-platform linux --python-version 3.11 \
#     requirements.in -o requirements.txt
#   uv pip compile --generate-hashes --overrides overrides.txt \
#     --python-platform linux --python-version 3.11 \
#     environments/requirements-lock.in -o environments/requirements.lock
#
# EXIT CONDITION: drop this override once torch raises its `setuptools<82` cap
# upstream, then re-run the two commands above without --overrides.
setuptools>=83.0.0
